Avg Handover Time Time Formatted = 
/*Measure returns the minutes and seconds (hh:mm:ss) for average Handover time. The average minutes which are then rounded down is used to return the minutes, then the rounded avg mins is subtracted from the total avg time to leave the seconds. The seconds are multiplied by 60 to provide the ss time format, and is then rounded. These are concatinated together using a colon delimeter to return the mm:ss string. */

VAR _Hrs =  ROUNDDOWN(AVERAGE('dwh FactAmbulance'[HandoverDurationSeconds])/3600,0)
VAR _mins = ROUNDDOWN(AVERAGE('dwh FactAmbulance'[HandoverDurationSeconds])/60,0) - (_Hrs*60)
VAR _seconds =
    ROUND(
        (AVERAGE(
            'dwh FactAmbulance'[HandoverDurationSeconds]) - (_mins*60) - (_Hrs*3600)
        ),0)
       
RETURN
IF(
    ISBLANK(_Hrs)&&ISBLANK(_mins)&&ISBLANK(_seconds), BLANK(),
 FORMAT(_Hrs,"00") & ":" & FORMAT(_mins,"00") & ":" & FORMAT(_seconds,"00")
)
